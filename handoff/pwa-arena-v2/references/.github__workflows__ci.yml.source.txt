  secret-scan:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1  # v7.0.1
        with:
          fetch-depth: 0
      # Use the MIT Gitleaks CLI, not gitleaks/gitleaks-action (its repository
      # does not declare a licence). Verify the official release checksum and
      # redact findings so a discovered credential is never printed to logs.
      - name: Install verified Gitleaks CLI
        run: |
          set -euo pipefail
          version=8.30.1
          archive="gitleaks_${version}_linux_x64.tar.gz"
          base="https://github.com/gitleaks/gitleaks/releases/download/v${version}"
          curl -fsSL "$base/$archive" -o "/tmp/$archive"
          curl -fsSL "$base/gitleaks_${version}_checksums.txt" -o /tmp/gitleaks-checksums.txt
          (cd /tmp && grep "  $archive$" gitleaks-checksums.txt | sha256sum --check)
          mkdir -p /tmp/gitleaks
          tar -xzf "/tmp/$archive" -C /tmp/gitleaks gitleaks
          /tmp/gitleaks/gitleaks version
      - name: Verify exact signing exceptions with synthetic secrets
        env:
          GITLEAKS: /tmp/gitleaks/gitleaks
        run: python3 scripts/test_retired_signing_scan.py
      - name: Scan full Git history (redacted metadata)
        run: |
          set +e
          /tmp/gitleaks/gitleaks git --redact --no-banner --log-level=error \
            --report-format=json --report-path=/tmp/gitleaks-report.json .
          scan_status=$?
          set -e
          python - <<'PY'
          import json
          import os
          from pathlib import Path

          report = Path("/tmp/gitleaks-report.json")
          findings = []
          if report.is_file():
              try:
                  parsed = json.loads(report.read_text(encoding="utf-8"))
                  findings = parsed if isinstance(parsed, list) else []
              except (OSError, json.JSONDecodeError):
                  findings = []

          def escape_command_value(value: object) -> str:
              return (
                  str(value)
                  .replace("%", "%25")
                  .replace("\r", "%0D")
                  .replace("\n", "%0A")
                  .replace(":", "%3A")
                  .replace(",", "%2C")
              )

          summary = ["### Gitleaks scan (redacted metadata)"]
          summary.append(f"Potential findings: {len(findings)}. Secret values are never included.")
          for finding in findings[:40]:
              path = finding.get("File", "unknown path")
              rule = finding.get("RuleID", "unknown rule")
              line = finding.get("StartLine", 0)
              commit = str(finding.get("Commit", ""))[:12]
              message = f"Potential secret match ({rule}); value withheld; commit {commit or 'unknown'}."
              print(
                  f"::error file={escape_command_value(path)},line={line},title=Gitleaks finding::"
                  f"{escape_command_value(message)}"
              )
              summary.append(f"- `{path}:{line}` — `{rule}`; commit `{commit or 'unknown'}` (value withheld).")
          if not report.is_file():
              summary.append("No JSON report was produced; inspect the scanner step output after checking access restrictions.")
              print("::error::Gitleaks failed without a readable JSON report; no secret values were emitted by this summary.")
          summary_path = os.environ.get("GITHUB_STEP_SUMMARY")
          if summary_path:
              with open(summary_path, "a", encoding="utf-8") as stream:
                  stream.write("\n".join(summary) + "\n")
          PY
          exit "$scan_status"

          import re
          import sys
          from pathlib import Path

          analyze_status = int(os.environ["ANALYZE_STATUS"])
          machine_status = int(os.environ["MACHINE_STATUS"])
          raw_lines = Path("/tmp/analyze.txt").read_text(encoding="utf-8", errors="replace").splitlines()
          error_lines = [line for line in raw_lines if re.match(r"^\s*error\s*•", line)]
          pattern = re.compile(r"^\s*error\s*•\s*(.*?)\s*•\s*(.+?):(\d+):(\d+)\s*•\s*(.*)$")
          machine_diagnostics = []
          for line in Path("/tmp/analyze.machine").read_text(encoding="utf-8", errors="replace").splitlines():
              fields = line.split("|", 7)
              if len(fields) == 8 and fields[0] in {"ERROR", "WARNING", "INFO"}:
                  severity, _kind, code, path, line_number, column, _length, message = fields
                  machine_diagnostics.append((severity, code, path, line_number, column, message))
          severity_order = {"ERROR": 0, "WARNING": 1, "INFO": 2}
          machine_diagnostics.sort(
              key=lambda item: (severity_order[item[0]], item[2], int(item[3]), int(item[4]))
          )

          def escape(value: str) -> str:
              return value.replace("%", "%25").replace("\r", "%0D").replace("\n", "%0A").replace(":", "%3A").replace(",", "%2C")

          summary = ["### Flutter analyzer diagnostics"]
          if analyze_status == 0:
              print("analyze: no fatal analyzer diagnostics")
              summary.append("No fatal analyzer diagnostics found.")
          elif machine_diagnostics:
              level = {"ERROR": "error", "WARNING": "warning", "INFO": "notice"}
              print(f"flutter analyze exited {analyze_status}; Dart machine analyzer status {machine_status}.")
              for severity, code, path, line_number, column, message in machine_diagnostics[:50]:
                  annotation = level[severity]
                  safe_message = f"{message} [{code}]" if code else message
                  print(
                      f"::{annotation} file={escape(path)},line={line_number},col={column},title=Dart analyzer::"
                      f"{escape(safe_message)}"
                  )
                  summary.append(f"- `{path}:{line_number}:{column}` — {message} (`{code}`; {severity.lower()})")
              summary.insert(1, f"Machine-format diagnostics: {len(machine_diagnostics)}.")
          elif error_lines:
              print("::error::flutter analyze: error diagnostics found; human-format fallback parsed")
              for line in error_lines[:50]:
                  match = pattern.match(line)
                  if match:
                      message, path, line_number, column, code = match.groups()
                      safe_message = f"{message} [{code}]" if code else message
                      print(
                          f"::error file={escape(path)},line={line_number},col={column},title=Dart analyzer::"
                          f"{escape(safe_message)}"
                      )
                      summary.append(f"- `{path}:{line_number}:{column}` — {message} (`{code}`)")
                  else:
                      print("::error::Flutter analyzer reported an error; see the redacted diagnostic artifact.")
                      summary.append("An analyzer error was reported; its location could not be parsed safely.")
              summary.insert(1, f"Error diagnostics: {len(error_lines)}.")
          else:
              print(f"::error::flutter analyze exited {analyze_status} without a parseable diagnostic")
              summary.append(
                  f"Analyzer command failed with exit status {analyze_status}; Dart machine analysis returned {machine_status} without parseable diagnostics."
              )
          summary.insert(1, f"Flutter analyzer exit status: {analyze_status}.")

              with open(summary_path, "a", encoding="utf-8") as stream:
