// ============================================
// THREE.JS INLINE (Minimal Custom 3D Engine)

(function() {
    'use strict';

    // ---- PRELOADER ----
    const preloader = document.getElementById('preloader');
    const preloaderBar = document.getElementById('preloader-bar');
    const preloaderPercent = document.getElementById('preloader-percent');
    let loadProgress = 0;

    function updatePreloader() {
        loadProgress += Math.random() * 15 + 5;
        if (loadProgress > 100) loadProgress = 100;
        preloaderBar.style.width = loadProgress + '%';
        preloaderPercent.textContent = Math.floor(loadProgress) + '%';

        if (loadProgress < 100) {
            setTimeout(updatePreloader, 150 + Math.random() * 200);
        } else {
            setTimeout(() => {
                preloader.classList.add('hidden');
                initScene();
            }, 500);
        }
    }
    setTimeout(updatePreloader, 300);

    // ---- CUSTOM CURSOR ----
    const cursorMain = document.getElementById('cursor-main');
    const cursorTrail = document.getElementById('cursor-trail');
    let mouseX = 0, mouseY = 0;
    let trailX = 0, trailY = 0;

    document.addEventListener('mousemove', (e) => {
        mouseX = e.clientX;
        mouseY = e.clientY;
        cursorMain.style.left = mouseX - 4 + 'px';
        cursorMain.style.top = mouseY - 4 + 'px';
    });

    function animateCursor() {
        trailX += (mouseX - trailX) * 0.15;
        trailY += (mouseY - trailY) * 0.15;
        cursorTrail.style.left = trailX - 20 + 'px';
        cursorTrail.style.top = trailY - 20 + 'px';
        requestAnimationFrame(animateCursor);
    }
    animateCursor();

    // Hover effects
    document.querySelectorAll('[data-hover]').forEach(el => {
        el.addEventListener('mouseenter', () => cursorTrail.classList.add('hovering'));
        el.addEventListener('mouseleave', () => cursorTrail.classList.remove('hovering'));
    });

    // ---- NAVIGATION ----
    const nav = document.getElementById('nav');
    window.addEventListener('scroll', () => {
        if (window.scrollY > 50) {
            nav.classList.add('scrolled');
        } else {
            nav.classList.remove('scrolled');
        }
    });

    // ---- SCROLL REVEAL ----
    const revealElements = document.querySelectorAll('.reveal, .celestial-card, .feature-card, .timeline-item');

    const revealObserver = new IntersectionObserver((entries) => {
        entries.forEach(entry => {
            if (entry.isIntersecting) {
                const delay = entry.target.dataset.delay || 0;
                setTimeout(() => {
                    entry.target.classList.add('visible');
                }, delay);
            }
        });
    }, { threshold: 0.1, rootMargin: '0px 0px -50px 0px' });

    revealElements.forEach(el => revealObserver.observe(el));

    // ---- COUNTER ANIMATION ----
    const counterElements = document.querySelectorAll('[data-count]');
    const counterObserver = new IntersectionObserver((entries) => {
        entries.forEach(entry => {
            if (entry.isIntersecting) {
                const target = parseFloat(entry.target.dataset.count);
                const suffix = entry.target.querySelector('.stat-suffix');
                const suffixText = suffix ? suffix.textContent : '';
                const isFloat = target % 1 !== 0;
                let current = 0;
                const duration = 2000;
                const step = target / (duration / 16);

                function count() {
                    current += step;
                    if (current >= target) {
                        current = target;
                        entry.target.innerHTML = (isFloat ? current.toFixed(1) : Math.floor(current)) + 
                            (suffixText ? '<span class="stat-suffix">' + suffixText + '</span>' : '');
                        return;
                    }
                    entry.target.innerHTML = (isFloat ? current.toFixed(1) : Math.floor(current)) + 
                        (suffixText ? '<span class="stat-suffix">' + suffixText + '</span>' : '');
                    requestAnimationFrame(count);
                }
                count();
                counterObserver.unobserve(entry.target);
            }
        });
    }, { threshold: 0.5 });

    counterElements.forEach(el => counterObserver.observe(el));

    // ---- TOOLTIP ----
    const tooltip = document.getElementById('tooltip');
    const tooltipTitle = document.getElementById('tooltip-title');
    const tooltipText = document.getElementById('tooltip-text');

    document.querySelectorAll('.info-panel-item').forEach(item => {
        item.addEventListener('mouseenter', (e) => {
            tooltipTitle.textContent = item.dataset.tooltipTitle;
            tooltipText.textContent = item.dataset.tooltipText;
            tooltip.classList.add('show');
        });
        item.addEventListener('mousemove', (e) => {
            tooltip.style.left = (e.clientX - 270) + 'px';
            tooltip.style.top = (e.clientY - 20) + 'px';
        });
        item.addEventListener('mouseleave', () => {
            tooltip.classList.remove('show');
        });
    });

    // ---- INFO PANEL ----
    document.querySelectorAll('.info-panel-item').forEach(item => {
        item.addEventListener('click', function() {
            document.querySelectorAll('.info-panel-item').forEach(i => i.classList.remove('active'));
            this.classList.add('active');
        });
    });

    // ---- SMOOTH SCROLL ----
    document.querySelectorAll('a[href^="#"]').forEach(anchor => {
        anchor.addEventListener('click', function(e) {
            e.preventDefault();
            const target = document.querySelector(this.getAttribute('href'));
            if (target) {
                target.scrollIntoView({ behavior: 'smooth', block: 'start' });
            }
        });
    });

    // ============================================
    // MAIN 3D SCENE - WebGL Custom Engine
    // ============================================

    function initScene() {
        const container = document.getElementById('canvas-container');
        const canvas = document.createElement('canvas');
        container.appendChild(canvas);
        const gl = canvas.getContext('webgl2') || canvas.getContext('webgl');
        if (!gl) {
            console.warn('WebGL not supported, falling back to 2D');
            init2DFallback(container);
            return;
        }

        let width = window.innerWidth;
        let height = window.innerHeight;
        canvas.width = width * window.devicePixelRatio;
        canvas.height = height * window.devicePixelRatio;
        canvas.style.width = width + 'px';
        canvas.style.height = height + 'px';

        // Resize
        window.addEventListener('resize', () => {
            width = window.innerWidth;
            height = window.innerHeight;
            canvas.width = width * window.devicePixelRatio;
            canvas.height = height * window.devicePixelRatio;
            canvas.style.width = width + 'px';
            canvas.style.height = height + 'px';
            gl.viewport(0, 0, canvas.width, canvas.height);
        });

        gl.viewport(0, 0, canvas.width, canvas.height);
        gl.enable(gl.BLEND);
        gl.blendFunc(gl.SRC_ALPHA, gl.ONE);

        // ---- Shader compilation ----
        function createShader(type, source) {
            const shader = gl.createShader(type);
            gl.shaderSource(shader, source);
            gl.compileShader(shader);
            if (!gl.getShaderParameter(shader, gl.COMPILE_STATUS)) {
                console.error(gl.getShaderInfoLog(shader));
                gl.deleteShader(shader);
                return null;
            }
            return shader;
        }

        function createProgram(vsSource, fsSource) {
            const vs = createShader(gl.VERTEX_SHADER, vsSource);
            const fs = createShader(gl.FRAGMENT_SHADER, fsSource);
            const program = gl.createProgram();
            gl.attachShader(program, vs);
            gl.attachShader(program, fs);
            gl.linkProgram(program);
            if (!gl.getProgramParameter(program, gl.LINK_STATUS)) {
                console.error(gl.getProgramInfoLog(program));
                return null;
            }
            return program;
        }

        // ---- STAR FIELD SHADER ----
        const starVS = `
            attribute vec3 aPosition;
            attribute float aSize;
            attribute float aBrightness;
            attribute vec3 aColor;
            
            uniform mat4 uProjection;
            uniform mat4 uView;
            uniform float uTime;
            uniform float uScrollY;
            
            varying float vBrightness;
            varying vec3 vColor;
            
            void main() {
                vec3 pos = aPosition;
                pos.z += uScrollY * 0.0005;
                pos.z = mod(pos.z + 2.0, 4.0) - 2.0;
                
                vec4 mvPos = uView * vec4(pos, 1.0);
                gl_Position = uProjection * mvPos;
                
                float twinkle = sin(uTime * (1.0 + aBrightness * 3.0) + aBrightness * 100.0) * 0.3 + 0.7;
                gl_PointSize = aSize * twinkle * (300.0 / -mvPos.z);
                
                vBrightness = aBrightness * twinkle;
                vColor = aColor;
            }
        `;

        const starFS = `
            precision mediump float;
            varying float vBrightness;
            varying vec3 vColor;
            
            void main() {
                vec2 center = gl_PointCoord - 0.5;
                float dist = length(center);
                if (dist > 0.5) discard;
                
                float alpha = smoothstep(0.5, 0.0, dist);
                alpha *= alpha;
                
                gl_FragColor = vec4(vColor * vBrightness, alpha * vBrightness);
            }
        `;

        // ---- BLACK HOLE EFFECT (fullscreen quad) ----
        const quadVS = `
            attribute vec2 aPosition;
            varying vec2 vUV;
            void main() {
                vUV = aPosition * 0.5 + 0.5;
                gl_Position = vec4(aPosition, 0.0, 1.0);
            }
        `;

        const blackholeFS = `
            precision highp float;
            varying vec2 vUV;
            uniform float uTime;
            uniform vec2 uResolution;
            uniform vec2 uMouse;
            uniform float uScrollY;
            
            #define PI 3.14159265359
            
            float hash(vec2 p) {
                return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453123);
            }
            
            float noise(vec2 p) {
                vec2 i = floor(p);
                vec2 f = fract(p);
                f = f * f * (3.0 - 2.0 * f);
                return mix(
                    mix(hash(i), hash(i + vec2(1.0, 0.0)), f.x),
                    mix(hash(i + vec2(0.0, 1.0)), hash(i + vec2(1.0, 1.0)), f.x),
                    f.y
                );
            }
            
            float fbm(vec2 p) {
                float val = 0.0;
                float amp = 0.5;
                for (int i = 0; i < 5; i++) {
                    val += amp * noise(p);
                    p *= 2.0;
                    amp *= 0.5;
                }
                return val;
            }
            
            void main() {
                vec2 uv = (gl_FragCoord.xy - 0.5 * uResolution) / min(uResolution.x, uResolution.y);
                
                // Mouse influence
                vec2 mouseUV = (uMouse - 0.5) * 2.0;
                mouseUV.x *= uResolution.x / uResolution.y;
                
                // Black hole position influenced by scroll
                vec2 bhPos = vec2(0.3 + sin(uTime * 0.1) * 0.05, 0.1 + cos(uTime * 0.08) * 0.05);
                bhPos.y -= uScrollY * 0.0003;
                
                vec2 toBH = uv - bhPos;
                float dist = length(toBH);
                float angle = atan(toBH.y, toBH.x);
                
                // Gravitational lensing
                float schwarzschild = 0.08;
                float lensStrength = schwarzschild / (dist * dist + 0.01);
                vec2 lensed = uv + normalize(toBH) * lensStrength * 0.02;
                
                // Accretion disk
                float diskAngle = angle + uTime * 0.5 + lensStrength * 2.0;
                float diskRadius = dist;
                float diskWidth = 0.15;
                float diskCenter = 0.25;
                float disk = smoothstep(diskWidth, 0.0, abs(diskRadius - diskCenter));
                
                // Disk detail
                float diskNoise = fbm(vec2(diskAngle * 3.0, diskRadius * 10.0 - uTime * 0.3));
                disk *= diskNoise * 1.5;
                
                // Doppler effect - disk colors
                float doppler = sin(diskAngle) * 0.5 + 0.5;
                vec3 diskColorHot = vec3(1.0, 0.6, 0.1);
                vec3 diskColorCold = vec3(0.4, 0.1, 0.8);
                vec3 diskColor = mix(diskColorCold, diskColorHot, doppler);
                
                // Inner glow
                float innerGlow = exp(-dist * 8.0) * 0.5;
                vec3 innerColor = vec3(0.5, 0.2, 1.0);
                
                // Event horizon (dark center)
                float eventHorizon = smoothstep(0.06, 0.04, dist);
                
                // Pulsar beams
                vec2 pulsarPos = vec2(-0.4, -0.2 + sin(uTime * 0.15) * 0.1);
                pulsarPos.y -= uScrollY * 0.0002;
                vec2 toPulsar = uv - pulsarPos;
                float pulsarDist = length(toPulsar);
                float pulsarAngle = atan(toPulsar.y, toPulsar.x);
                
                // Rotating beam
                float beamAngle = uTime * 3.0;
                float beam1 = pow(max(0.0, cos(pulsarAngle - beamAngle)), 100.0);
                float beam2 = pow(max(0.0, cos(pulsarAngle - beamAngle + PI)), 100.0);
                float beams = (beam1 + beam2) * exp(-pulsarDist * 2.0);
                vec3 beamColor = vec3(0.1, 0.8, 1.0);
                
                // Pulsar core
                float pulsarCore = exp(-pulsarDist * 20.0);
                float pulsarPulse = sin(uTime * 10.0) * 0.3 + 0.7;
                
                // Nebula background
                vec2 nebulaUV = uv * 2.0 + uTime * 0.02;
                float nebula = fbm(nebulaUV) * fbm(nebulaUV + 3.0);
                vec3 nebulaColor1 = vec3(0.1, 0.0, 0.2);
                vec3 nebulaColor2 = vec3(0.0, 0.05, 0.15);
                vec3 nebulaMix = mix(nebulaColor1, nebulaColor2, fbm(nebulaUV * 0.5));
                
                // Compose
                vec3 color = nebulaMix * nebula * 0.3;
                
                // Add accretion disk
                color += diskColor * disk * (1.0 - eventHorizon) * 1.5;
                
                // Add inner glow
                color += innerColor * innerGlow * (1.0 - eventHorizon);
                
                // Add photon sphere
                float photonSphere = smoothstep(0.01, 0.0, abs(dist - 0.07)) * 0.8;
                color += vec3(0.8, 0.5, 1.0) * photonSphere * (1.0 - eventHorizon);
                
                // Apply event horizon
                color *= (1.0 - eventHorizon * 0.98);
                
                // Add pulsar
                color += beamColor * beams * 0.8;
                color += vec3(0.3, 0.9, 1.0) * pulsarCore * pulsarPulse;
                
                // Add subtle mouse-reactive glow
                float mouseDist = length(uv - mouseUV * 0.3);
                color += vec3(0.2, 0.1, 0.4) * exp(-mouseDist * 4.0) * 0.15;
                
                // Vignette
                float vignette = 1.0 - dot(uv * 0.5, uv * 0.5);
                color *= max(0.0, vignette);
                
                // Tone mapping
                color = color / (color + 1.0);
                color = pow(color, vec3(0.85));
                
                gl_FragColor = vec4(color, 1.0);
            }
        `;

        // Create programs
        const starProgram = createProgram(starVS, starFS);
        const bhProgram = createProgram(quadVS, blackholeFS);

        // ---- Generate Stars ----
        const STAR_COUNT = 3000;
        const starPositions = new Float32Array(STAR_COUNT * 3);
        const starSizes = new Float32Array(STAR_COUNT);
        const starBrightness = new Float32Array(STAR_COUNT);
        const starColors = new Float32Array(STAR_COUNT * 3);

        for (let i = 0; i < STAR_COUNT; i++) {
            starPositions[i * 3] = (Math.random() - 0.5) * 6;
            starPositions[i * 3 + 1] = (Math.random() - 0.5) * 6;
            starPositions[i * 3 + 2] = (Math.random() - 0.5) * 4;
            
            starSizes[i] = Math.random() * 3 + 0.5;
            starBrightness[i] = Math.random();
            
            // Star color temperature
            const temp = Math.random();
            if (temp < 0.3) { // Blue-white
                starColors[i * 3] = 0.7 + Math.random() * 0.3;
                starColors[i * 3 + 1] = 0.8 + Math.random() * 0.2;
                starColors[i * 3 + 2] = 1.0;
            } else if (temp < 0.7) { // White-yellow
                starColors[i * 3] = 1.0;
                starColors[i * 3 + 1] = 0.9 + Math.random() * 0.1;
                starColors[i * 3 + 2] = 0.7 + Math.random() * 0.3;
            } else { // Orange-red
                starColors[i * 3] = 1.0;
                starColors[i * 3 + 1] = 0.5 + Math.random() * 0.3;
                starColors[i * 3 + 2] = 0.3 + Math.random() * 0.2;
            }
        }

        // Star buffers
        const starPosBuffer = gl.createBuffer();
        gl.bindBuffer(gl.ARRAY_BUFFER, starPosBuffer);
        gl.bufferData(gl.ARRAY_BUFFER, starPositions, gl.STATIC_DRAW);

        const starSizeBuffer = gl.createBuffer();
        gl.bindBuffer(gl.ARRAY_BUFFER, starSizeBuffer);
        gl.bufferData(gl.ARRAY_BUFFER, starSizes, gl.STATIC_DRAW);

        const starBrightnessBuffer = gl.createBuffer();
        gl.bindBuffer(gl.ARRAY_BUFFER, starBrightnessBuffer);
        gl.bufferData(gl.ARRAY_BUFFER, starBrightness, gl.STATIC_DRAW);

        const starColorBuffer = gl.createBuffer();
        gl.bindBuffer(gl.ARRAY_BUFFER, starColorBuffer);
        gl.bufferData(gl.ARRAY_BUFFER, starColors, gl.STATIC_DRAW);

        // Fullscreen quad buffer
        const quadBuffer = gl.createBuffer();
        gl.bindBuffer(gl.ARRAY_BUFFER, quadBuffer);
        gl.bufferData(gl.ARRAY_BUFFER, new Float32Array([-1,-1, 1,-1, -1,1, 1,1]), gl.STATIC_DRAW);

        // ---- Matrix utilities ----
        function perspective(fov, aspect, near, far) {
            const f = 1.0 / Math.tan(fov / 2);
            return new Float32Array([
                f / aspect, 0, 0, 0,
                0, f, 0, 0,
                0, 0, (far + near) / (near - far), -1,
                0, 0, (2 * far * near) / (near - far), 0
            ]);
        }

        function lookAt(eye, target, up) {
            const zAxis = normalize(subtract(eye, target));
            const xAxis = normalize(cross(up, zAxis));
            const yAxis = cross(zAxis, xAxis);
            return new Float32Array([
                xAxis[0], yAxis[0], zAxis[0], 0,
                xAxis[1], yAxis[1], zAxis[1], 0,
                xAxis[2], yAxis[2], zAxis[2], 0,
                -dot(xAxis, eye), -dot(yAxis, eye), -dot(zAxis, eye), 1
            ]);
        }

        function normalize(v) {
            const len = Math.sqrt(v[0]*v[0] + v[1]*v[1] + v[2]*v[2]);
            return [v[0]/len, v[1]/len, v[2]/len];
        }

        function subtract(a, b) { return [a[0]-b[0], a[1]-b[1], a[2]-b[2]]; }
        function cross(a, b) { return [a[1]*b[2]-a[2]*b[1], a[2]*b[0]-a[0]*b[2], a[0]*b[1]-a[1]*b[0]]; }
        function dot(a, b) { return a[0]*b[0] + a[1]*b[1] + a[2]*b[2]; }

        // ---- Animation state ----
        let scrollY = 0;
        let targetScrollY = 0;
        let normalizedMouseX = 0.5;
        let normalizedMouseY = 0.5;
        let targetMouseX = 0.5;
        let targetMouseY = 0.5;

        window.addEventListener('scroll', () => {
            targetScrollY = window.scrollY;
        });

        document.addEventListener('mousemove', (e) => {
            targetMouseX = e.clientX / window.innerWidth;
            targetMouseY = 1.0 - e.clientY / window.innerHeight;
        });

        // ---- Render Loop ----
        let startTime = performance.now();

        function render() {
            const time = (performance.now() - startTime) * 0.001;

            // Smooth interpolation
            scrollY += (targetScrollY - scrollY) * 0.05;
            normalizedMouseX += (targetMouseX - normalizedMouseX) * 0.05;
            normalizedMouseY += (targetMouseY - normalizedMouseY) * 0.05;

            gl.viewport(0, 0, canvas.width, canvas.height);
            gl.clearColor(0, 0, 0, 1);
            gl.clear(gl.COLOR_BUFFER_BIT);

            const aspect = canvas.width / canvas.height;

            // ---- Draw Black Hole Background ----
            gl.disable(gl.BLEND);
            gl.useProgram(bhProgram);

            const bhPosLoc = gl.getAttribLocation(bhProgram, 'aPosition');
            gl.bindBuffer(gl.ARRAY_BUFFER, quadBuffer);
            gl.enableVertexAttribArray(bhPosLoc);
            gl.vertexAttribPointer(bhPosLoc, 2, gl.FLOAT, false, 0, 0);

            gl.uniform1f(gl.getUniformLocation(bhProgram, 'uTime'), time);
            gl.uniform2f(gl.getUniformLocation(bhProgram, 'uResolution'), canvas.width, canvas.height);
            gl.uniform2f(gl.getUniformLocation(bhProgram, 'uMouse'), normalizedMouseX, normalizedMouseY);
            gl.uniform1f(gl.getUniformLocation(bhProgram, 'uScrollY'), scrollY);

            gl.drawArrays(gl.TRIANGLE_STRIP, 0, 4);
            gl.disableVertexAttribArray(bhPosLoc);

            // ---- Draw Stars ----
            gl.enable(gl.BLEND);
            gl.blendFunc(gl.SRC_ALPHA, gl.ONE);
            gl.useProgram(starProgram);

            // Camera
            const camX = Math.sin(time * 0.05) * 0.3 + (normalizedMouseX - 0.5) * 0.5;
            const camY = Math.cos(time * 0.03) * 0.2 + (normalizedMouseY - 0.5) * 0.3;
            const camZ = 2.0 - scrollY * 0.0003;

            const projMatrix = perspective(Math.PI / 3, aspect, 0.1, 100.0);
            const viewMatrix = lookAt([camX, camY, camZ], [0, 0, 0], [0, 1, 0]);

            gl.uniformMatrix4fv(gl.getUniformLocation(starProgram, 'uProjection'), false, projMatrix);
            gl.uniformMatrix4fv(gl.getUniformLocation(starProgram, 'uView'), false, viewMatrix);
            gl.uniform1f(gl.getUniformLocation(starProgram, 'uTime'), time);
            gl.uniform1f(gl.getUniformLocation(starProgram, 'uScrollY'), scrollY);

            // Position
            const posLoc = gl.getAttribLocation(starProgram, 'aPosition');
            gl.bindBuffer(gl.ARRAY_BUFFER, starPosBuffer);
            gl.enableVertexAttribArray(posLoc);
            gl.vertexAttribPointer(posLoc, 3, gl.FLOAT, false, 0, 0);

            // Size
            const sizeLoc = gl.getAttribLocation(starProgram, 'aSize');
            gl.bindBuffer(gl.ARRAY_BUFFER, starSizeBuffer);
            gl.enableVertexAttribArray(sizeLoc);
            gl.vertexAttribPointer(sizeLoc, 1, gl.FLOAT, false, 0, 0);

            // Brightness
            const brightLoc = gl.getAttribLocation(starProgram, 'aBrightness');
            gl.bindBuffer(gl.ARRAY_BUFFER, starBrightnessBuffer);
            gl.enableVertexAttribArray(brightLoc);
            gl.vertexAttribPointer(brightLoc, 1, gl.FLOAT, false, 0, 0);

            // Color
            const colorLoc = gl.getAttribLocation(starProgram, 'aColor');
            gl.bindBuffer(gl.ARRAY_BUFFER, starColorBuffer);
            gl.enableVertexAttribArray(colorLoc);
            gl.vertexAttribPointer(colorLoc, 3, gl.FLOAT, false, 0, 0);

            gl.drawArrays(gl.POINTS, 0, STAR_COUNT);

            gl.disableVertexAttribArray(posLoc);
            gl.disableVertexAttribArray(sizeLoc);
            gl.disableVertexAttribArray(brightLoc);
            gl.disableVertexAttribArray(colorLoc);

            requestAnimationFrame(render);
        }

        render();
    }

    // ---- Black Hole Card Canvas ----
    function initBlackHoleCard() {
        const canvas = document.getElementById('blackhole-canvas');
        if (!canvas) return;
        const ctx = canvas.getContext('2d');

        function resize() {
            const rect = canvas.parentElement.getBoundingClientRect();
            canvas.width = rect.width * window.devicePixelRatio;
            canvas.height = rect.height * window.devicePixelRatio;
            canvas.style.width = rect.width + 'px';
            canvas.style.height = rect.height + 'px';
        }
        resize();
        window.addEventListener('resize', resize);

        let cardMouseX = 0.5, cardMouseY = 0.5;
        canvas.parentElement.addEventListener('mousemove', (e) => {
            const rect = canvas.parentElement.getBoundingClientRect();
            cardMouseX = (e.clientX - rect.left) / rect.width;
            cardMouseY = (e.clientY - rect.top) / rect.height;
        });

        // Particles orbiting black hole
        const particles = [];
        const PARTICLE_COUNT = 200;
        for (let i = 0; i < PARTICLE_COUNT; i++) {
            const angle = Math.random() * Math.PI * 2;
            const radius = 0.15 + Math.random() * 0.35;
            particles.push({
                angle: angle,
                radius: radius,
                speed: (0.3 + Math.random() * 0.7) / radius,
                size: Math.random() * 2 + 0.5,
                brightness: Math.random(),
                hue: Math.random() * 60 + 250 // Purple to blue range
            });
        }

        function drawBH(time) {
            const w = canvas.width;
            const h = canvas.height;
            const dpr = window.devicePixelRatio;
            const cx = w * (0.5 + (cardMouseX - 0.5) * 0.05);
            const cy = h * (0.5 + (cardMouseY - 0.5) * 0.05);

            ctx.clearRect(0, 0, w, h);

            // Black hole shadow
            const bhRadius = Math.min(w, h) * 0.08;
            const gradient = ctx.createRadialGradient(cx, cy, bhRadius * 0.5, cx, cy, bhRadius * 5);
            gradient.addColorStop(0, 'rgba(0,0,0,1)');
            gradient.addColorStop(0.2, 'rgba(0,0,0,0.95)');
            gradient.addColorStop(0.4, 'rgba(80,30,150,0.1)');
            gradient.addColorStop(0.7, 'rgba(40,10,100,0.02)');
            gradient.addColorStop(1, 'rgba(0,0,0,0)');
            ctx.fillStyle = gradient;
            ctx.fillRect(0, 0, w, h);

            // Accretion disk glow
            ctx.save();
            ctx.translate(cx, cy);
            ctx.scale(1, 0.3);
            
            for (let ring = 3; ring > 0; ring--) {
                const ringRadius = bhRadius * (1.5 + ring * 0.8);
                const ringGrad = ctx.createRadialGradient(0, 0, ringRadius - 10 * dpr, 0, 0, ringRadius + 10 * dpr);
                
                const hue = 270 + ring * 20 + time * 10;
                ringGrad.addColorStop(0, `hsla(${hue}, 80%, 60%, 0)`);
                ringGrad.addColorStop(0.4, `hsla(${hue}, 80%, 60%, ${0.15 / ring})`);
                ringGrad.addColorStop(0.5, `hsla(${hue}, 90%, 70%, ${0.3 / ring})`);
                ringGrad.addColorStop(0.6, `hsla(${hue}, 80%, 60%, ${0.15 / ring})`);
                ringGrad.addColorStop(1, `hsla(${hue}, 80%, 60%, 0)`);
                
                ctx.fillStyle = ringGrad;
                ctx.beginPath();
                ctx.arc(0, 0, ringRadius + 15 * dpr, 0, Math.PI * 2);
                ctx.fill();
            }
            ctx.restore();

            // Orbiting particles
            particles.forEach(p => {
                p.angle += p.speed * 0.01;
                const x = cx + Math.cos(p.angle) * p.radius * w;
                const y = cy + Math.sin(p.angle) * p.radius * h * 0.3;
                
                const distToCenterNorm = p.radius;
                const alpha = p.brightness * (0.3 + Math.sin(time * 2 + p.angle) * 0.2);
                
                ctx.fillStyle = `hsla(${p.hue + time * 20}, 80%, ${60 + p.brightness * 30}%, ${alpha})`;
                ctx.beginPath();
                ctx.arc(x, y, p.size * dpr, 0, Math.PI * 2);
                ctx.fill();
            });

            // Photon ring
            ctx.save();
            ctx.translate(cx, cy);
            ctx.scale(1, 0.3);
            ctx.strokeStyle = `rgba(200, 150, 255, ${0.2 + Math.sin(time * 3) * 0.1})`;
            ctx.lineWidth = 1.5 * dpr;
            ctx.beginPath();
            ctx.arc(0, 0, bhRadius * 1.3, 0, Math.PI * 2);
            ctx.stroke();
            ctx.restore();

            // Center black
            ctx.fillStyle = '#000';
            ctx.beginPath();
            ctx.arc(cx, cy, bhRadius * 0.6, 0, Math.PI * 2);
            ctx.fill();
        }

        let bhTime = 0;
        function animateBH() {
            bhTime += 0.016;
            drawBH(bhTime);
            requestAnimationFrame(animateBH);
        }

        // Start when visible
        const bhObserver = new IntersectionObserver((entries) => {
            if (entries[0].isIntersecting) {
                animateBH();
                bhObserver.unobserve(canvas);
            }
        }, { threshold: 0.1 });
        bhObserver.observe(canvas);
    }

    // Initialize BH card when DOM ready
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', initBlackHoleCard);
    } else {
        initBlackHoleCard();
    }

    // ---- 2D Fallback ----
    function init2DFallback(container) {
        const canvas = document.createElement('canvas');
        container.appendChild(canvas);
        const ctx = canvas.getContext('2d');

        function resize() {
            canvas.width = window.innerWidth * window.devicePixelRatio;
            canvas.height = window.innerHeight * window.devicePixelRatio;
            canvas.style.width = window.innerWidth + 'px';
            canvas.style.height = window.innerHeight + 'px';
        }
        resize();
        window.addEventListener('resize', resize);

        const stars = [];
        for (let i = 0; i < 500; i++) {
            stars.push({
                x: Math.random(),
                y: Math.random(),
                size: Math.random() * 2 + 0.5,
                brightness: Math.random(),
                speed: Math.random() * 0.0002 + 0.0001
            });
        }

        function draw() {
            const w = canvas.width;
            const h = canvas.height;
            ctx.fillStyle = '#000';
            ctx.fillRect(0, 0, w, h);

            stars.forEach(s => {
                s.y += s.speed;
                if (s.y > 1) s.y = 0;
                
                const twinkle = Math.sin(performance.now() * 0.002 * s.brightness) * 0.3 + 0.7;
                ctx.fillStyle = `rgba(255, 255, 255, ${s.brightness * twinkle})`;
                ctx.beginPath();
                ctx.arc(s.x * w, s.y * h, s.size * window.devicePixelRatio, 0, Math.PI * 2);
                ctx.fill();
            });

            requestAnimationFrame(draw);
        }
        draw();
    }

})();
